-- Mr.Scripts Hub - Pro UI Injection (Fixed Main Area)
-- discord.gg/K4xE6Q5yFb

local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local CoreGui = game:GetService("CoreGui")
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")
local TweenService = game:GetService("TweenService")

-- تحميل المكتبة
local libSrc = game:HttpGet("https://pastefy.app/LVKkOffd/raw")
local Lib = loadstring(libSrc)()

-- إنشاء النافذة بالاسم والديسكورد
local Window = Lib:CreateWindow({
    Title      = "Mr.Scripts",
    SubTitle   = "Hub Scripts",
    Theme      = "Dark",
    Discord    = "discord.gg/K4xE6Q5yFb",
    Ambient    = "particles",
    Scale      = 1,
})

task.wait(0.5) -- انتظار بسيط عشان الواجهة تتحمل بالكامل

-- البحث عن الـ ScreenGui
local sg
local coreGui = gethui and gethui() or CoreGui
for _, v in ipairs(coreGui:GetDescendants()) do
    if v:IsA("ScreenGui") and v.DisplayOrder == 999999 then
        sg = v
        break
    end
end
if not sg then
    for _, v in ipairs(PlayerGui:GetDescendants()) do
        if v:IsA("ScreenGui") and v.DisplayOrder == 999999 then
            sg = v
            break
        end
    end
end

if not sg then return end

-- البحث عن المنطقة الرئيسية (Frame19) اللي في النص
local mainPageContainer
for _, v in ipairs(sg:GetDescendants()) do
    -- القائمة الجانبية عرضها بيكون صغير (أقل من 100)
    if v:IsA("ScrollingFrame") and v.AbsoluteSize.X > 0 and v.AbsoluteSize.X < 100 then
        local frame23 = v.Parent and v.Parent.Parent
        if frame23 then
            -- بندور على الفريم اللي جنبها اللي هو منطقة المحتوى الرئيسية
            for _, child in ipairs(frame23:GetChildren()) do
                if child:IsA("Frame") and child.Position.X.Offset == 62 then
                    mainPageContainer = child
                    break
                end
            end
        end
        break
    end
end

-- لو ملقتش بالطريقة دي، بندور على أكبر فريم في الواجهة
if not mainPageContainer then
    local maxX = 0
    for _, v in ipairs(sg:GetDescendants()) do
        if v:IsA("Frame") and v.Position.X.Offset == 62 and v.AbsoluteSize.X > maxX then
            maxX = v.AbsoluteSize.X
            mainPageContainer = v
        end
    end
end

if mainPageContainer then
    -- مسح أي حقن قديم لو السكريبت اتشغل أكتر من مرة
    local old = mainPageContainer:FindFirstChild("MrScriptContainer")
    if old then old:Destroy() end

    -- إنشاء منطقة العرض الجديدة جوه المنطقة الكبيرة
    local newScroll = Instance.new("ScrollingFrame")
    newScroll.Name = "MrScriptContainer"
    newScroll.Size = UDim2.new(1, 0, 1, 0)
    newScroll.BackgroundTransparency = 1
    newScroll.BorderSizePixel = 0
    newScroll.ScrollBarThickness = 4
    newScroll.ScrollBarImageColor3 = Color3.fromRGB(190, 70, 255)
    newScroll.CanvasSize = UDim2.new(0, 0, 0, 0)
    newScroll.AutomaticCanvasSize = Enum.AutomaticSize.Y
    newScroll.Parent = mainPageContainer

    local padding = Instance.new("UIPadding")
    padding.PaddingTop = UDim.new(0, 15)
    padding.PaddingLeft = UDim.new(0, 15)
    padding.PaddingRight = UDim.new(0, 15)
    padding.Parent = newScroll

    local layout = Instance.new("UIListLayout")
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Padding = UDim.new(0, 10)
    layout.Parent = newScroll

    -- عنوان القسم
    local sectionTitle = Instance.new("TextLabel")
    sectionTitle.Size = UDim2.new(1, 0, 0, 30)
    sectionTitle.BackgroundTransparency = 1
    sectionTitle.Font = Enum.Font.BuilderSansExtraBold
    sectionTitle.TextSize = 14
    sectionTitle.TextColor3 = Color3.fromRGB(190, 70, 255)
    sectionTitle.TextXAlignment = Enum.TextXAlignment.Left
    sectionTitle.Text = "AVAILABLE SCRIPTS"
    sectionTitle.LayoutOrder = 0
    sectionTitle.Parent = newScroll

    -- دالة إنشاء الكروت
    local function CreateScriptCard(titleText, descText, hasInput, loadFunction)
        local card = Instance.new("Frame")
        card.Name = "MrScriptCard"
        card.Size = UDim2.new(1, -10, 0, 65)
        card.BackgroundColor3 = Color3.fromRGB(28, 23, 41)
        card.BorderSizePixel = 0
        card.ClipsDescendants = true
        card.Parent = newScroll

        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, 10)
        corner.Parent = card

        local stroke = Instance.new("UIStroke")
        stroke.Color = Color3.fromRGB(190, 70, 255)
        stroke.Transparency = 1
        stroke.Thickness = 1.5
        stroke.Parent = card

        local topSection = Instance.new("TextButton")
        topSection.Size = UDim2.new(1, 0, 0, 65)
        topSection.BackgroundTransparency = 1
        topSection.Text = ""
        topSection.AutoButtonColor = false
        topSection.Parent = card
        
        local title = Instance.new("TextLabel")
        title.Size = UDim2.new(1, -20, 0, 24)
        title.Position = UDim2.fromOffset(15, 10)
        title.BackgroundTransparency = 1
        title.Font = Enum.Font.BuilderSansExtraBold
        title.TextSize = 16
        title.TextColor3 = Color3.fromRGB(236, 233, 245)
        title.TextXAlignment = Enum.TextXAlignment.Left
        title.Text = titleText
        title.Parent = topSection

        local desc = Instance.new("TextLabel")
        desc.Size = UDim2.new(1, -20, 0, 16)
        desc.Position = UDim2.fromOffset(15, 32)
        desc.BackgroundTransparency = 1
        desc.Font = Enum.Font.BuilderSansMedium
        desc.TextSize = 12
        desc.TextColor3 = Color3.fromRGB(139, 132, 160)
        desc.TextXAlignment = Enum.TextXAlignment.Left
        desc.Text = descText
        desc.Parent = topSection

        local inputSection
        if hasInput then
            inputSection = Instance.new("Frame")
            inputSection.Size = UDim2.new(1, -30, 0, 40)
            inputSection.Position = UDim2.fromOffset(15, 70)
            inputSection.BackgroundTransparency = 1
            inputSection.Visible = false
            inputSection.Parent = card

            local textBox = Instance.new("TextBox")
            textBox.Size = UDim2.new(1, -90, 1, 0)
            textBox.Position = UDim2.fromOffset(0, 0)
            textBox.BackgroundColor3 = Color3.fromRGB(18, 15, 27)
            textBox.BorderSizePixel = 0
            textBox.Font = Enum.Font.BuilderSansMedium
            textBox.TextSize = 14
            textBox.TextColor3 = Color3.fromRGB(236, 233, 245)
            textBox.PlaceholderText = "Enter Robux amount..."
            textBox.PlaceholderColor3 = Color3.fromRGB(139, 132, 160)
            textBox.Text = ""
            textBox.ClearTextOnFocus = false
            textBox.Parent = inputSection

            local tbCorner = Instance.new("UICorner")
            tbCorner.CornerRadius = UDim.new(0, 8)
            tbCorner.Parent = textBox

            local tbStroke = Instance.new("UIStroke")
            tbStroke.Color = Color3.fromRGB(190, 70, 255)
            tbStroke.Transparency = 0.5
            tbStroke.Parent = textBox

            local loadBtn = Instance.new("TextButton")
            loadBtn.Size = UDim2.fromOffset(80, 0)
            loadBtn.Position = UDim2.new(1, -80, 0, 0)
            loadBtn.BackgroundColor3 = Color3.fromRGB(190, 70, 255)
            loadBtn.BorderSizePixel = 0
            loadBtn.Font = Enum.Font.BuilderSansExtraBold
            loadBtn.TextSize = 14
            loadBtn.TextColor3 = Color3.new(1, 1, 1)
            loadBtn.Text = "LOAD"
            loadBtn.Parent = inputSection

            local btnGradient = Instance.new("UIGradient")
            btnGradient.Color = ColorSequence.new(Color3.fromRGB(190, 70, 255), Color3.fromRGB(138, 43, 226))
            btnGradient.Rotation = 90
            btnGradient.Parent = loadBtn

            local btnCorner = Instance.new("UICorner")
            btnCorner.CornerRadius = UDim.new(0, 8)
            btnCorner.Parent = loadBtn
            
            loadBtn.MouseButton1Click:Connect(function()
                local amount = tonumber(textBox.Text)
                if amount and amount > 0 then
                    loadBtn.Text = "..."
                    loadFunction(math.floor(amount))
                    task.wait(0.5)
                    inputSection.Visible = false
                    TweenService:Create(card, TweenInfo.new(0.3, Enum.EasingStyle.Quart), {Size = UDim2.new(1, -10, 0, 65)}):Play()
                    TweenService:Create(stroke, TweenInfo.new(0.2), {Transparency = 1}):Play()
                    loadBtn.Text = "LOAD"
                else
                    textBox.Text = ""
                    textBox.PlaceholderText = "Invalid Number!"
                end
            end)
        end

        topSection.MouseEnter:Connect(function()
            TweenService:Create(stroke, TweenInfo.new(0.2), {Transparency = 0.3}):Play()
            TweenService:Create(card, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(36, 29, 53)}):Play()
        end)
        
        topSection.MouseLeave:Connect(function()
            if not (inputSection and inputSection.Visible) then
                TweenService:Create(stroke, TweenInfo.new(0.2), {Transparency = 1}):Play()
            end
            TweenService:Create(card, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(28, 23, 41)}):Play()
        end)

        topSection.MouseButton1Click:Connect(function()
            if hasInput then
                inputSection.Visible = not inputSection.Visible
                if inputSection.Visible then
                    TweenService:Create(card, TweenInfo.new(0.3, Enum.EasingStyle.Quart), {Size = UDim2.new(1, -10, 0, 120)}):Play()
                    TweenService:Create(stroke, TweenInfo.new(0.2), {Transparency = 0}):Play()
                else
                    TweenService:Create(card, TweenInfo.new(0.3, Enum.EasingStyle.Quart), {Size = UDim2.new(1, -10, 0, 65)}):Play()
                    TweenService:Create(stroke, TweenInfo.new(0.2), {Transparency = 1}):Play()
                end
            else
                loadFunction()
            end
        end)
    end

    -- 1. Perm Magnet
    CreateScriptCard("Perm Magnet", "Set Robux amount, then load perm magnet", true, function(amount)
        getgenv().robux = amount
        pcall(function()
            loadstring(game:HttpGet("https://raw.githubusercontent.com/vOlOrex/magpr/refs/heads/main/1"))()
        end)
    end)

    -- 2. Perm Dragon
    CreateScriptCard("Perm Dragon", "Set Robux amount, then load perm dragon", true, function(amount)
        getgenv().robux = amount
        pcall(function()
            loadstring(game:HttpGet("https://raw.githubusercontent.com/vOlOrex/drgpr/refs/heads/main/12"))()
        end)
    end)

    -- 3. 100x Luck
    CreateScriptCard("100x Luck", "Click to instantly load the script", false, function()
        pcall(function()
            loadstring(game:HttpGet("https://raw.githubusercontent.com/vOlOrex/1xlc/refs/heads/main/12"))()
        end)
    end)

    -- 4. Fruit Spawner
    CreateScriptCard("Fruit Spawner", "Click to open the spawner menu", false, function()
        pcall(function()
            loadstring(game:HttpGet("https://raw.githubusercontent.com/vOlOrex/spwn/refs/heads/main/22"))()
        end)
    end)
end
